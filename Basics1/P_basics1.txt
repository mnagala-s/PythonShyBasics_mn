import pandas as pd

df = pd.read_csv("protein_results.csv")

# Clean data
df = df.dropna(subset=["Sample_ID", "Protein", "Concentration_ng_ml"])

# Add a signal classification
df["Signal_Class"] = df["Concentration_ng_ml"].apply(
    lambda x: "High" if x >= 2 else "Low"
)

# Calculate mean and SD across replicates
summary = (
    df.groupby(["Sample_ID", "Protein"], as_index=False)
      .agg(
          Mean_ng_ml=("Concentration_ng_ml", "mean"),
          SD_ng_ml=("Concentration_ng_ml", "std"),
          N=("Concentration_ng_ml", "count")
      )
)

# Save output
summary.to_excel("protein_assay_summary.xlsx", index=False)

print(summary)
