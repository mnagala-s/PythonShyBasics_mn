1. Smallest Shiny app

from shiny import App, ui, render

app_ui = ui.page_fluid(
    ui.h1("My first Shiny app"),
    ui.input_text("name", "Enter your name", value="Manjula"),
    ui.output_text("greeting")
)

def server(input, output, session):

    @render.text
    def greeting():
        return f"Hello, {input.name()}!"

app = App(app_ui, server)

##############################################

Run it:

shiny run --reload app.py

input.name() reads the current value of the UI item whose ID is "name".ui.output_text("greeting") reserves a page location, and the function named greeting() supplies its content.

3. Reactive calculationsUse @reactive.calc for a calculation that depends on inputs and can be reused by more than one output.

from shiny import App, ui, render, reactive
import pandas as pd

df1 = pd.read_csv("df_proteomics_logData.csv")

app_ui = ui.page_fluid(
    ui.input_slider(
        "minimum_fc",
        "Minimum absolute log2 fold change",
        min=0,
        max=5,
        value=1,
        step=0.1
    ),

    ui.output_text("protein_count"),
    ui.output_data_frame("protein_table")
)

def server(input, output, session):

    @reactive.calc
    def filtered_data():
        return df1[
            df1["log2_FoldChange_Treated_vs_Control"].abs()
            >= input.minimum_fc()
        ].copy()

    @render.text
    def protein_count():
        return f"Proteins passing filter: {len(filtered_data())}"

    @render.data_frame
    def protein_table():
        return render.DataGrid(filtered_data(), filters=True)

app = App(app_ui, server)

import pandas as pd

df1 = pd.read_csv("df_proteomics_logData.csv")

app_ui = ui.page_fluid(
    ui.input_slider(
        "minimum_fc",
        "Minimum absolute log2 fold change",
        min=0,
        max=5,
        value=1,
        step=0.1
    ),

    ui.output_text("protein_count"),
    ui.output_data_frame("protein_table")
)

def server(input, output, session):

    @reactive.calc
    def filtered_data():
        return df1[
            df1["log2_FoldChange_Treated_vs_Control"].abs()
            >= input.minimum_fc()
        ].copy()

    @render.text
    def protein_count():
        return f"Proteins passing filter: {len(filtered_data())}"

    def protein_table():

app = App(app_ui, server)

#The important point is that whenever the slider changes, filtered_data() is recalculated automatically. Both the count and table then update.
